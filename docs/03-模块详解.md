```python
def run_single_task(self, index):
    self.apply_config_to_env()  # 确保配置最新
    
    task = self.tasks[index]
    cls = SCRIPT_REGISTRY.get(task["type"])
    if not cls: return "未知任务类型"
    
    try:
        instance = cls(task["params"])  # 实例化
        instance.run()                  # 执行
        return f"[{task['name']}] 执行完毕"
    except Exception as e:
        return f"执行出错: {e}"
```

#### 批量运行任务
```python
def run_all_tasks(self):
    self.apply_config_to_env()
    
    # 获取循环次数
    total_loops = int(self.config.get("loop_count", "1"))
    
    for cycle in range(total_loops):
        print(f">>> 正在执行第 {cycle + 1} / {total_loops} 轮 ...")
        
        for i, t in enumerate(self.tasks):
            if t["enable"]:
                self.run_single_task(i)
        
        # 轮次间休息
        if cycle < total_loops - 1:
            print(">>> 本轮结束，等待 5 秒...")
            time.sleep(5)
    
    return f"全部完成！共执行 {total_loops} 轮。"
```

```python
CURRENT_DEVICE_ID = None  # 全局变量，缓存设备ID

def auto_select_device():
    global CURRENT_DEVICE_ID
    
    if CURRENT_DEVICE_ID:
        return CURRENT_DEVICE_ID
    
    # 1. 检查已连接设备
    device_id = check_devices()
    if device_id:
        CURRENT_DEVICE_ID = device_id
        return device_id
    
    # 2. 尝试连接常见端口

        CURRENT_DEVICE_ID = device_id

    return None
```

#### 执行ADB命令
```python
def run_adb_cmd(cmd_list):
    adb_path = get_adb_path()
    device_id = auto_select_device()
    
    # 构建命令：adb -s <device_id> <cmd>
    base_cmd = [adb_path]
    if device_id:
        base_cmd.extend(["-s", device_id])
    
    final_cmd = base_cmd + cmd_list
    
    result = subprocess.run(final_cmd, stdout=subprocess.PIPE, timeout=15)
    return result.stdout.decode('utf-8', errors='ignore')
```

### 3.4 高级操作

#### 连续拖拽 (Monkey脚本)

**核心原理**：
- Android的`input swipe`命令只能实现单段滑动
- 要实现连续拖拽，需要使用Monkey脚本
- Monkey脚本可以精确控制触摸事件序列

**实现**：
```python
def run_continuous_drag(point_list, hold_time=1200):
    # 参数配置
    STEPS_PER_SEGMENT = 40   # 插值点数量
    STEP_WAIT_MS = 5         # 每步等待时间
    
    # 构建脚本
    script_content = "type= user\ncount= 10\nspeed= 1.0\nstart data >>\n"
    
    # 1. 按下
    start_x, start_y = point_list[0]
    script_content += f"DispatchPointer(0, 0, 0, {start_x}, {start_y}, ...)\n"
    
    # 2. 等待（抓取）
    script_content += f"UserWait({hold_time})\n"
    
    # 3. 移动（插值）
    for i in range(len(point_list) - 1):
        p1, p2 = point_list[i], point_list[i+1]
        
        for step in range(1, STEPS_PER_SEGMENT + 1):
            t = step / STEPS_PER_SEGMENT
            x = int(p1[0] + (p2[0] - p1[0]) * t)
            y = int(p1[1] + (p2[1] - p1[1]) * t)
            
            script_content += f"DispatchPointer(0, 0, 2, {x}, {y}, ...)\n"
            script_content += f"UserWait({STEP_WAIT_MS})\n"
    
    # 4. 抬起
    end_x, end_y = point_list[-1]
    script_content += f"DispatchPointer(0, 0, 1, {end_x}, {end_y}, ...)\n"
    
    # 写入文件并执行
    with open("temp_drag.mks", "w") as f:
        f.write(script_content)
    
    subprocess.run([adb_path, "-s", device_id, "push", "temp_drag.mks", "/sdcard/"])
    subprocess.run([adb_path, "-s", device_id, "shell", "monkey", "-f", "/sdcard/temp_drag.mks", "1"])
```

### 3.5 命令解析器

#### execute_multiline_adb()

支持的指令格式：
```
# 注释
sleep 1.5
input tap 100 200
input swipe 100 200 300 400 500
drag_path 100 100 200 200 300 300
```

**实现**：
```python
def execute_multiline_adb(text_block):
    lines = text_block.strip().split('\n')
    
    for line in lines:
        line = line.strip()
        if not line or line.startswith("#"): 
            continue
        
        parts = line.split()
        cmd = parts[0].lower()
        
        # 处理 sleep
        if cmd in ["sleep", "wait", "timeout"]:
            time.sleep(float(parts[1]))
            continue
        
        # 处理 drag_path
        if cmd == "drag_path":
            coords = []
            raw_nums = [int(x) for x in parts[1:]]
            coords = list(zip(raw_nums[::2], raw_nums[1::2]))
            run_continuous_drag(coords)
            continue
        
        # 处理普通ADB命令
        if line.startswith("adb "):
            line = line[4:]
        
        adb_args = ["shell"] + line.split()
        run_adb_cmd(adb_args)
        time.sleep(0.1)
```

```python
def stitch_images(img_path_left, img_path_right, output_path, 
                  split_x_left=None, split_x_right=None):
    img1 = Image.open(img_path_left)
    img2 = Image.open(img_path_right)
    w, h = img1.size
    
    # 裁剪
    cut_l = int(split_x_left) if split_x_left else w
    cut_r = int(split_x_right) if split_x_right else 0
    
    part1 = img1.crop((0, 0, cut_l, h))
    part2 = img2.crop((cut_r, 0, w, h))
    
    # 拼接
    new_w = part1.width + part2.width
    new_img = Image.new('RGB', (new_w, h))
    new_img.paste(part1, (0, 0))
    new_img.paste(part2, (part1.width, 0))
    
    new_img.save(output_path)
```

## 4. AI客户端模块 (ai_client.py)

### 4.1 客户端管理

#### 每次调用创建新客户端
```python
def get_client():
    api_key = os.environ.get("OPENAI_API_KEY")
    base_url = os.environ.get("ai_base_url", 
                              "https://open.bigmodel.cn/api/paas/v4")
    
    if not api_key:
        return None
    
    # 每次实例化都会重新计算JWT签名
    return OpenAI(api_key=api_key, base_url=base_url)
```

**为什么每次创建？**
- 智谱AI使用JWT Token认证
- Token有时效性（通常1小时）
- 每次创建新客户端会重新生成Token
- 避免长时间运行后Token过期导致401错误

### 4.2 图片处理

#### 压缩和编码
```python
def convert_image_to_webp_base64(input_image_path, quality=80):
    with Image.open(input_image_path) as img:
        # 缩放（如果太大）
        if img.width > 1024 or img.height > 1024:
            img.thumbnail((1024, 1024))
        
        # 转WebP
        byte_arr = io.BytesIO()
        img.save(byte_arr, format='WEBP', quality=quality)
        
        # Base64编码
        return base64.b64encode(byte_arr.getvalue()).decode('utf-8')
```

**优化点**：
- **缩放**: 减少Token消耗，加快识别速度
- **WebP格式**: 比PNG/JPG体积更小，质量相当
- **质量80%**: 在大小和质量间取得平衡

```python
def query_vlm(image_path, prompt, model='glm-4v-flash'):
    # 1. 获取客户端
    client = get_client()
    if not client:
        return "错误：API Key 未设置"
    
    # 2. 构建消息
    messages = build_messages(image_path, prompt)
    if not messages:
        return "错误：图片转换失败"
    
    # 3. 发起请求
    response = client.chat.completions.create(
        model=model,
        messages=messages,
        stream=False,
        temperature=0.1,    # 低温度，减少随机性
        max_tokens=1024
    )
    
    result = response.choices[0].message.content
    return result
```

```python
def reset_state(self):
    # 1. 侧滑唤醒
    adb_swipe(600, 500, 800, 500, 300)
    time.sleep(self.s_delay)
    
    # 2. 点击订单
    pos_order = parse_coordinate(os.environ.get("reset_pos_order"))
    if pos_order:
        adb_click(pos_order[0], pos_order[1])
    time.sleep(self.s_delay)
    
    # 3. 缩放
    adb_zoom_out()
    time.sleep(self.s_delay + 1.0)
    
    # 4. 退出订单
    pos_exit_order = parse_coordinate(os.environ.get("reset_pos_exit_order"))
    if pos_exit_order:
        adb_click(pos_exit_order[0], pos_exit_order[1])
    time.sleep(self.s_delay)
    
    # 5. 点击蒲公英
    pos_dandelion = parse_coordinate(os.environ.get("reset_pos_dandelion"))
    if pos_dandelion:
        adb_click(pos_dandelion[0], pos_dandelion[1])
    time.sleep(self.b_delay)
    
    # 6. 退出蒲公英
```
