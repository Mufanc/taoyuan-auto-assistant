# 开发指南

本文档介绍如何扩展和定制功能。

## 1. 添加新任务类型

### 1.1 基本步骤

1. 在`src/tasks.py`中创建新类
2. 继承`GameScriptBase`
3. 定义`LABEL`和`PARAM_CONFIG`
4. 实现`execute()`方法
5. 添加到`SCRIPT_REGISTRY`

### 1.2 完整示例

```python
class FishingTask(GameScriptBase):
    """钓鱼任务示例"""
    
    # 1. 定义标签（会显示在UI的任务类型下拉框中）
    LABEL = "7. 钓鱼 (自动版)"
    
    # 2. 定义参数配置
    PARAM_CONFIG = {
        # 进入钓鱼场的ADB命令
        "area_adb": {
            "label": "进入钓鱼场ADB",
            "type": "text",  # 多行文本框
            "default": "input tap 1000 500\nsleep 2"
        },
        
        # 鱼竿位置
        "rod_pos": {
            "label": "鱼竿位置坐标 (x y)",
            "type": "string",  # 单行输入框
            "default": "800 400"
        },
        
        # 钓鱼次数
        "fish_count": {
            "label": "钓鱼次数",
            "type": "int",  # 单行输入框（会转为整数）
            "default": "10"
        },
        
        # 退出钓鱼场
        "quit_adb": {
            "label": "退出钓鱼场ADB",
            "type": "text",
            "default": "input tap 100 100\nsleep 1"
        }
    }
    
    # 3. 实现核心逻辑
    def execute(self):
        """具体执行逻辑"""
        self.log("开始钓鱼...")
        
        # 读取参数
        rod_pos = parse_coordinate(self.params.get("rod_pos"))
        fish_count = int(self.params.get("fish_count", 10))
        
        # 钓鱼循环
        for i in range(fish_count):
            self.log(f"第 {i+1}/{fish_count} 次钓鱼")
            
            # 1. 点击鱼竿
            if rod_pos:
                adb_click(rod_pos[0], rod_pos[1])
                time.sleep(2)
            
            # 2. 截图识别状态
            img_path = "fishing_state.png"
            adb_screenshot(img_path)
            
            prompt = "判断是否出现'收竿'按钮，返回yes或no"
            result = query_vlm(img_path, prompt)
            
            # 3. 如果可以收竿
            if "yes" in result.lower():
                self.log("检测到鱼上钩，收竿！")
                adb_click(rod_pos[0], rod_pos[1])  # 再次点击收竿
                time.sleep(self.b_delay)  # 等待收竿动画
            else:
                self.log("未检测到鱼上钩，等待...")
                time.sleep(5)
        
        self.log("钓鱼完成！")

# 4. 注册到任务表
SCRIPT_REGISTRY = {
    t.LABEL: t for t in [
        FarmingTask,
        GatheringTask,
        ProcessingTask,
        CookingTask,
        OrderTask,
        DandelionTask,
        FishingTask  # 添加新任务
    ]
}
```

### 1.3 参数配置说明

#### type字段
- `"text"`: 多行文本框，用于ADB命令
- `"string"`: 单行输入框，用于坐标、名称等
- `"int"`: 单行输入框，会被转换为整数

#### label字段
- 显示在UI上的参数名称
- 建议使用中文，清晰易懂

#### default字段
- 默认值
- 新建任务时会自动填充

## 2. 添加新的ADB操作

### 2.1 在adb_utils.py中添加函数

```python
def adb_long_press(x, y, duration=2000):
    """长按操作（通过swipe实现）"""
    run_adb_cmd(["shell", "input", "swipe", 
                 str(x), str(y), str(x), str(y), str(duration)])

def adb_input_text(text):
    """输入文本"""
    # 注意：需要转义特殊字符
    text = text.replace(" ", "%s")
    run_adb_cmd(["shell", "input", "text", text])

def adb_press_key(keycode):
    """按键操作
    常用keycode:
    - KEYCODE_BACK = 4
    - KEYCODE_HOME = 3
    - KEYCODE_MENU = 82
    """
    run_adb_cmd(["shell", "input", "keyevent", str(keycode)])

def adb_get_screen_size():
    """获取屏幕尺寸"""
    output = run_adb_cmd(["shell", "wm", "size"])
    # 输出格式: Physical size: 1600x900
    import re
    match = re.search(r'(\d+)x(\d+)', output)
    if match:
        return int(match.group(1)), int(match.group(2))
    return None
```

### 2.2 在execute_multiline_adb中添加支持

```python
def execute_multiline_adb(text_block):
    # ... 现有代码 ...
    
    # 添加新指令支持
    if cmd == "long_press":
        # 格式: long_press x y duration
        x, y = int(parts[1]), int(parts[2])
        duration = int(parts[3]) if len(parts) > 3 else 2000
        adb_long_press(x, y, duration)
        continue
    
    if cmd == "input_text":
        # 格式: input_text "Hello World"
        text = " ".join(parts[1:]).strip('"')
        adb_input_text(text)
        continue
    
    if cmd == "press_key":
        # 格式: press_key 4
        keycode = int(parts[1])
        adb_press_key(keycode)
        continue
```

## 5. 性能优化

### 5.1 减少截图次数

```python
# 不好的做法：重复截图
for i in range(10):
    adb_screenshot(f"item_{i}.png")
    # ...

# 好的做法：复用截图
adb_screenshot("items.png")
for i in range(10):
    # 使用同一张图或者裁剪
    # ...
```

### 5.2 使用裁剪截图

```python
# 全图截图（慢）
adb_screenshot("full.png")

# 裁剪截图（快）
crop_box = (1000, 400, 1200, 500)  # (x1, y1, x2, y2)
adb_screenshot("crop.png", crop_box=crop_box)
```

### 5.3 批量操作

```python
# 不好的做法：逐个处理
for pos in positions:
    adb_click(*pos)
    time.sleep(1)
    adb_screenshot(...)
    # ...

# 好的做法：批量收集数据后统一处理
screenshots = []
for i, pos in enumerate(positions):
    adb_click(*pos)
    time.sleep(0.5)
    screenshots.append(adb_screenshot(f"state_{i}.png"))

# 批量识别
results = [query_vlm(img, prompt) for img in screenshots]
```

### 5.4 调整延迟

```python
# 根据实际情况调整延迟
# 不需要等待的操作
time.sleep(0.1)

# 需要等待UI加载的操作
time.sleep(self.s_delay)  # 1.5秒

# 需要等待动画的操作
time.sleep(self.b_delay)  # 8秒
```

## 7. 常见问题解决

### 7.1 ADB连接失败

```python
# 问题：more than one device/emulator
# 原因：连接了多个设备

# 解决：查看设备列表
# adb devices

# 手动指定设备
# adb -s 127.0.0.1:16384 shell ...

# 或者在代码中设置
CURRENT_DEVICE_ID = "127.0.0.1:16384"
```

### 7.2 截图返回空白

```python
# 问题：截图失败或返回空白

# 检查：
# 1. 是否有截图权限
# 2. 是否使用了正确的adb路径
# 3. 设备是否正常连接

# 解决：重置设备ID
CURRENT_DEVICE_ID = None
auto_select_device()  # 重新检测
```

### 7.3 AI识别不准确

```python
# 问题：AI返回的结果不符合预期

# 优化方案：
# 1. 使用裁剪截图，只截取关键区域
crop_box = (x1, y1, x2, y2)
adb_screenshot("target.png", crop_box=crop_box)

# 2. 改进提示词，提供更多上下文
prompt = """
这是游戏中的库存数字显示。
图中只有一个数字，范围是0-999。
请识别这个数字，只返回数字本身。
"""

# 3. 数据清洗
result = query_vlm(...)
result = result.strip().replace('o', '0').replace('O', '0')

# 4. 正则提取
nums = re.findall(r'\d+', result)
value = int(nums[0]) if nums else 0
```

### 7.4 操作过快导致失败

```python
# 问题：点击后界面还没加载，就执行了下一步

# 解决：增加等待时间
adb_click(x, y)
time.sleep(self.s_delay)  # 等待UI加载

# 或者：使用AI识别等待状态
def wait_for_ui_ready(self, expected_state, timeout=10):
    """等待UI就绪"""
    start_time = time.time()
    
    while time.time() - start_time < timeout:
        adb_screenshot("wait_state.png")
        result = query_vlm("wait_state.png", "判断当前状态")
        
        if expected_state in result:
            return True
        
        time.sleep(0.5)
    
    return False
```

### 9.2 注释规范

```python
def execute(self):
    """
    执行采集任务
    
    流程：
    1. 循环遍历各个地点
    2. 识别当前状态
    3. 根据状态执行对应操作
    """
    # 读取参数
    count = int(self.params.get("location_count", 1))
    
    # 主循环
    for i in range(count):
        # 处理单个地点
        self.process_location(i)
```

### 9.3 错误处理

```python
def execute(self):
    try:
        # 主逻辑
        result = self.do_something()
        
    except Exception as e:
        self.log(f"执行出错: {e}")
        # 可选：尝试恢复
        self.reset_state()
    
    finally:
        # 清理工作
        self.cleanup()
```
